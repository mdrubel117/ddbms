### Test Docker Run

### Download Hadoop Image

docker pull macio232/hadoop-pseudo-distributed-mode

### New Hadoop Container Create and Run 

docker run -p 9870:9870 -p 8088:8088 -it --name=myHadoop macio232/hadoop-pseudo-distributed-mode

## docker container start and stop (do another terminal)

docker container start -i  myHadoop 
docker stop myHadoop

### Create customer table ( in/bankdb/customer)
CREATE TABLE customer (
customer_id BIGINT,
name VARCHAR(100),
email VARCHAR(100)
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/bankdb/customer';

### create account table with partitioned column

 CREATE TABLE account (
  account_id BIGINT,
  customer_id BIGINT,
  balance DECIMAL(12,2)
)
PARTITIONED BY (
  branch_id INT  
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/bankdb/account';

### create transaction table with partitioned column
CREATE TABLE Transaction (
  transaction_id BIGINT,
  account_id BIGINT,
  amount DECIMAL(12,2)
)
PARTITIONED BY (
  branch_id INT  
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/bankdb/Transaction';

show tables;
